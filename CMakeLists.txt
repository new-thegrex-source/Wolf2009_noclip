cmake_minimum_required(VERSION 3.20)
project(WolfRetailFreecam LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Build with -A Win32: retail Wolf2.exe is x86")
endif()
add_library(WolfFreecam SHARED WolfFreecam.cpp)
target_compile_features(WolfFreecam PRIVATE cxx_std_17)
target_compile_options(WolfFreecam PRIVATE /W4 /utf-8)
target_link_libraries(WolfFreecam PRIVATE user32)
add_executable(WolfFreecamInjector Injector.cpp)
target_compile_features(WolfFreecamInjector PRIVATE cxx_std_17)
target_compile_options(WolfFreecamInjector PRIVATE /W4 /utf-8)
add_executable(Wolf2Enhaced WIN32 Launcher.cpp)
target_compile_features(Wolf2Enhaced PRIVATE cxx_std_17)
target_compile_options(Wolf2Enhaced PRIVATE /W4 /utf-8)
target_link_libraries(Wolf2Enhaced PRIVATE user32)
target_link_libraries(WolfFreecam PRIVATE gdi32)
add_executable(PortalModeTest EXCLUDE_FROM_ALL PortalModeTest.cpp)
target_compile_features(PortalModeTest PRIVATE cxx_std_17)
target_compile_options(PortalModeTest PRIVATE /W4 /utf-8)
target_link_libraries(PortalModeTest PRIVATE user32 gdi32)
target_link_libraries(Wolf2Enhaced PRIVATE gdi32)

set_target_properties(Wolf2Enhaced PROPERTIES OUTPUT_NAME Wolf2Enhanced)
